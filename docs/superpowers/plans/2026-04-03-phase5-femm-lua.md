### Task 1: Implement femm_semi_on_pcb generator

**Files:**

The generator takes the same config struct as `workflow_semi_on_pcb` and produces a Lua script following the pattern of `semiHeatflow.lua`.

Key mapping from config to FEMM geometry:
- `cfg.areaContact` → `rc = sqrt(areaContact/pi)` (source radius)
- `cfg.areaDissBottom` → `b = sqrt(areaDissBottom/pi)` (outer radius)
- `cfg.pcbLayerStack` — each row `[thickness, conductivity]` becomes a layer
- `cfg.sinkLayerStack` — heatsink layers below PCB
- `cfg.pLossJunction / cfg.areaContact` → heat flux BC at top
- `cfg.hFluidBottom`, `cfg.tempFluidBottom` → convection BC at bottom
- Optional `cfg.hFluidTop`, `cfg.tempFluidTop` → convection BC at top

```matlab
function lua_str = femm_semi_on_pcb(cfg)
    % femm_semi_on_pcb - generate FEMM Lua script for semi-on-PCB model
    % cfg: same struct as workflow_semi_on_pcb config
    % Returns: string containing complete Lua script

    L = {};  % cell array of lines
    L{end+1} = '-------------------------------------------------------';
    L{end+1} = '-- Auto-generated by thermal_cli.m gen-femm';
    L{end+1} = '-- Model: semi-on-pcb (axisymmetric)';
    L{end+1} = sprintf('-- Date: %s', datestr(now, 'yyyy-mm-dd'));
    L{end+1} = '-------------------------------------------------------';
    L{end+1} = '';
    L{end+1} = 'newdocument(2)';
    L{end+1} = 'hi_probdef("meters","axi",1e-8,20,30)';
    L{end+1} = '';

    % Collect unique materials from layer stacks
    L{end+1} = '-- Materials';
    materials = {};
    mat_k = [];
    pcb = cfg.pcbLayerStack;
    for i = 1:size(pcb, 1)
        k = pcb(i, 2);
        name = sprintf('layer_pcb_%d', i);
        if size(pcb, 2) >= 3
            kip = pcb(i, 3);
        else
            kip = k;
        end
        L{end+1} = sprintf('hi_addmaterial("%s",%.6g,%.6g,0)', name, k, kip);
        materials{end+1} = name;
        mat_k(end+1) = k;
    end
    if isfield(cfg, 'sinkLayerStack') && ~isempty(cfg.sinkLayerStack)
        sink = cfg.sinkLayerStack;
        for i = 1:size(sink, 1)
            k = sink(i, 2);
            name = sprintf('layer_sink_%d', i);
            if size(sink, 2) >= 3
                kip = sink(i, 3);
            else
                kip = k;
            end
            L{end+1} = sprintf('hi_addmaterial("%s",%.6g,%.6g,0)', name, k, kip);
        end
    end
    L{end+1} = '';

    % Geometry parameters
    rc = sqrt(cfg.areaContact / pi);
    b = sqrt(cfg.areaDissBottom / pi);
    q = -sum(cfg.pLossJunction) / cfg.areaContact;  % heat flux (negative = into domain)
    eps = 0.001e-3;  % small offset for block labels

    L{end+1} = '-- Geometry parameters';
    L{end+1} = sprintf('rc = %.6e', rc);
    L{end+1} = sprintf('b = %.6e', b);
    L{end+1} = sprintf('q = %.6e', q);
    L{end+1} = sprintf('eps = %.6e', eps);
    L{end+1} = '';

    % Boundary conditions
    L{end+1} = '-- Boundary conditions';
    L{end+1} = sprintf('hi_addboundprop("heatflux",1,0,q)');
    L{end+1} = sprintf('hi_addboundprop("convection_bot",2,0,0,%.6f,%.6f)', ...
        cfg.tempFluidBottom, cfg.hFluidBottom);
    if cfg.includeTop
        L{end+1} = sprintf('hi_addboundprop("convection_top",2,0,0,%.6f,%.6f)', ...
            cfg.tempFluidTop, cfg.hFluidTop);
    else
        L{end+1} = 'hi_addboundprop("insulated_top",1,0,0)';
    end
    L{end+1} = '';

    % Build geometry: layers stack downward from y=0
    L{end+1} = '-- Geometry: nodes and segments';
    y_coords = [0];
    layer_names = {};

    % PCB layers
    y = 0;
    for i = 1:size(pcb, 1)
        y = y - pcb(i, 1);
        y_coords(end+1) = y;
        layer_names{end+1} = sprintf('layer_pcb_%d', i);
    end
    y_pcb_bottom = y;

    % Sink layers
    if isfield(cfg, 'sinkLayerStack') && ~isempty(cfg.sinkLayerStack)
        sink = cfg.sinkLayerStack;
        for i = 1:size(sink, 1)
            y = y - sink(i, 1);
            y_coords(end+1) = y;
            layer_names{end+1} = sprintf('layer_sink_%d', i);
        end
    end
    y_bottom = y;

    % Add nodes
    % At y=0: nodes at r=0, rc, b (top of PCB)
    % At intermediate y: nodes at rc, b (PCB layer boundaries)
    % At y_pcb_bottom: nodes at 0, rc, b
    % At y_bottom: nodes at 0, b

    % Top surface (y=0)
    L{end+1} = sprintf('hi_addnode(0, 0)');
    L{end+1} = sprintf('hi_addnode(%.6e, 0)', rc);
    L{end+1} = sprintf('hi_addnode(%.6e, 0)', b);

    % Intermediate PCB layers
    for i = 2:length(y_coords)-1
        if y_coords(i) == y_pcb_bottom && i < length(y_coords)
            L{end+1} = sprintf('hi_addnode(0, %.6e)', y_coords(i));
        end
        L{end+1} = sprintf('hi_addnode(%.6e, %.6e)', rc, y_coords(i));
        L{end+1} = sprintf('hi_addnode(%.6e, %.6e)', b, y_coords(i));
    end

    % Bottom surface
    L{end+1} = sprintf('hi_addnode(0, %.6e)', y_bottom);
    L{end+1} = sprintf('hi_addnode(%.6e, %.6e)', b, y_bottom);
    L{end+1} = '';

    % Segments: top surface
    L{end+1} = '-- Segments';
    L{end+1} = sprintf('hi_addsegment(0, 0, %.6e, 0)', rc);
    L{end+1} = sprintf('hi_addsegment(%.6e, 0, %.6e, 0)', rc, b);

    % Vertical segments and horizontal interfaces for each layer
    L{end+1} = sprintf('hi_addsegment(%.6e, 0, %.6e, %.6e)', b, b, y_coords(2));
    for i = 2:length(y_coords)-1
        L{end+1} = sprintf('hi_addsegment(%.6e, %.6e, %.6e, %.6e)', rc, y_coords(i), b, y_coords(i));
        if i < length(y_coords) - 1
            L{end+1} = sprintf('hi_addsegment(%.6e, %.6e, %.6e, %.6e)', rc, y_coords(i), rc, y_coords(i+1));
            L{end+1} = sprintf('hi_addsegment(%.6e, %.6e, %.6e, %.6e)', b, y_coords(i), b, y_coords(i+1));
        end
    end

    % Left boundary (axis of symmetry)
    L{end+1} = sprintf('hi_addsegment(0, 0, 0, %.6e)', y_pcb_bottom);

    % Bottom segments
    if y_bottom ~= y_pcb_bottom
        L{end+1} = sprintf('hi_addsegment(0, %.6e, 0, %.6e)', y_pcb_bottom, y_bottom);
        L{end+1} = sprintf('hi_addsegment(%.6e, %.6e, %.6e, %.6e)', b, y_coords(end-1), b, y_bottom);
    end
    L{end+1} = sprintf('hi_addsegment(0, %.6e, %.6e, %.6e)', y_bottom, b, y_bottom);
    L{end+1} = '';

    % Block labels and material assignment
    L{end+1} = '-- Block labels and material assignment';
    for i = 1:length(layer_names)
        y_mid = (y_coords(i) + y_coords(i+1)) / 2;
        x_mid = (rc + b) / 2;
        L{end+1} = sprintf('hi_addblocklabel(%.6e, %.6e)', x_mid, y_mid);
        L{end+1} = sprintf('hi_selectlabel(%.6e, %.6e)', x_mid, y_mid);
        L{end+1} = sprintf('hi_setblockprop("%s",0,0.05,0)', layer_names{i});
        L{end+1} = 'hi_clearselected()';
    end
    L{end+1} = '';

    % Apply boundary conditions
    L{end+1} = '-- Boundary conditions';
    L{end+1} = sprintf('hi_selectsegment(%.6e, 0)', rc/2);
    L{end+1} = 'hi_setsegmentprop("heatflux",0,1,0,0,"<None>")';
    L{end+1} = 'hi_clearselected()';

    if cfg.includeTop
        L{end+1} = sprintf('hi_selectsegment(%.6e, 0)', (rc+b)/2);
        L{end+1} = 'hi_setsegmentprop("convection_top",0,1,0,0,"<None>")';
        L{end+1} = 'hi_clearselected()';
    else
        L{end+1} = sprintf('hi_selectsegment(%.6e, 0)', (rc+b)/2);
        L{end+1} = 'hi_setsegmentprop("insulated_top",0,1,0,0,"<None>")';
        L{end+1} = 'hi_clearselected()';
    end

    L{end+1} = sprintf('hi_selectsegment(%.6e, %.6e)', b/2, y_bottom);
    L{end+1} = 'hi_setsegmentprop("convection_bot",0,1,0,0,"<None>")';
    L{end+1} = 'hi_clearselected()';
    L{end+1} = '';

    % Solve
    L{end+1} = '-- Solve';
    L{end+1} = 'hi_zoomnatural()';
    L{end+1} = 'hi_saveas("semi_on_pcb_model.feh")';
    L{end+1} = 'hi_analyze()';
    L{end+1} = 'hi_loadsolution()';
    L{end+1} = '';

    % Post-processing: extract to CSV
    L{end+1} = '-- Extract results to CSV';
    L{end+1} = 'f = openfile("femm_results.csv", "w")';
    L{end+1} = 'write(f, "point,x,y,temperature\\n")';
    L{end+1} = 'T_junction = ho_getpointvalues(0, 0)';
    L{end+1} = 'write(f, string.format("junction,0,0,%.6f\\n", T_junction))';
    L{end+1} = sprintf('T_pcb_bot = ho_getpointvalues(0, %.6e)', y_pcb_bottom);
    L{end+1} = sprintf('write(f, string.format("pcb_bottom,0,%.6e,%%.6f\\n", T_pcb_bot))', y_pcb_bottom);
    L{end+1} = sprintf('T_bottom = ho_getpointvalues(0, %.6e)', y_bottom);
    L{end+1} = sprintf('write(f, string.format("bottom,0,%.6e,%%.6f\\n", T_bottom))', y_bottom);
    L{end+1} = 'closefile(f)';
    L{end+1} = '';
    L{end+1} = 'print("FEMM simulation complete. Results in femm_results.csv")';

    lua_str = strjoin(L, '\n');
end
```

```matlab
function lua_str = femm_extruded_fin(cfg)
    % femm_extruded_fin - generate FEMM Lua script for extruded fin 2D model
    % 2D planar cross-section of one half-channel + half-fin
    % Symmetry exploited: model half-fin + half-channel

    L = {};
    L{end+1} = '-------------------------------------------------------';
    L{end+1} = '-- Auto-generated by thermal_cli.m gen-femm';
    L{end+1} = '-- Model: extruded-fin (2D planar cross-section)';
    L{end+1} = sprintf('-- Date: %s', datestr(now, 'yyyy-mm-dd'));
    L{end+1} = '-------------------------------------------------------';
    L{end+1} = '';
    L{end+1} = 'newdocument(2)';
    L{end+1} = 'hi_probdef("meters","planar",1e-8,20,30)';
    L{end+1} = '';

    % Geometry: half-fin + half-channel cross section
    % x: 0 = fin center, tf/2 = fin edge, tf/2 + wch/2 = channel center (symmetry)
    % y: 0 = base bottom, tb = base top / fin bottom, tb + Hf = fin top
    tf = cfg.heatsink.thickWall;  % fin thickness [m] (using thickWall as fin thickness)
    wch = cfg.heatsink.widthChannel;  % channel width [m]
    tb = cfg.heatsink.thickHeatsink;  % base thickness [m]
    Hf = cfg.heatsink.heightTotal - tb;  % fin height [m]
    k = cfg.heatsink.kSink;  % thermal conductivity [W/(m*K)]

    % Heat flux from component
    q_flux = cfg.heating.pLoss / (cfg.heating.widthContact * cfg.heating.lengthContact);

    % Convection on channel walls (from analytical result or estimate)
    if isfield(cfg, 'h_channel')
        h_ch = cfg.h_channel;
    else
        h_ch = 500;  % default estimate
    end
    T_fluid = cfg.fluid.tInlet;

    half_tf = tf / 2;
    half_wch = wch / 2;
    W = half_tf + half_wch;  % total width of model

    L{end+1} = '-- Materials';
    L{end+1} = sprintf('hi_addmaterial("heatsink_mat",%.6g,%.6g,0)', k, k);
    L{end+1} = '';

    L{end+1} = '-- Boundary conditions';
    L{end+1} = sprintf('hi_addboundprop("heat_flux",1,0,%.6e)', -q_flux);
    L{end+1} = sprintf('hi_addboundprop("convection",2,0,0,%.6f,%.6f)', T_fluid, h_ch);
    L{end+1} = 'hi_addboundprop("symmetry",1,0,0)';  % insulated = symmetry
    L{end+1} = '';

    % Nodes (cross-section)
    % Base: rectangle from (0,0) to (W, tb)
    % Fin: rectangle from (0, tb) to (half_tf, tb+Hf)
    L{end+1} = '-- Geometry nodes';
    L{end+1} = sprintf('hi_addnode(0, 0)');
    L{end+1} = sprintf('hi_addnode(%.6e, 0)', W);
    L{end+1} = sprintf('hi_addnode(0, %.6e)', tb);
    L{end+1} = sprintf('hi_addnode(%.6e, %.6e)', half_tf, tb);
    L{end+1} = sprintf('hi_addnode(%.6e, %.6e)', W, tb);
    L{end+1} = sprintf('hi_addnode(0, %.6e)', tb + Hf);
    L{end+1} = sprintf('hi_addnode(%.6e, %.6e)', half_tf, tb + Hf);
    L{end+1} = '';

    % Segments
    L{end+1} = '-- Segments';
    % Base bottom
    L{end+1} = sprintf('hi_addsegment(0, 0, %.6e, 0)', W);
    % Base right
    L{end+1} = sprintf('hi_addsegment(%.6e, 0, %.6e, %.6e)', W, W, tb);
    % Base top: fin part + channel part
    L{end+1} = sprintf('hi_addsegment(0, %.6e, %.6e, %.6e)', tb, half_tf, tb);
    L{end+1} = sprintf('hi_addsegment(%.6e, %.6e, %.6e, %.6e)', half_tf, tb, W, tb);
    % Base left (symmetry)
    L{end+1} = sprintf('hi_addsegment(0, 0, 0, %.6e)', tb);
    % Fin left (symmetry)
    L{end+1} = sprintf('hi_addsegment(0, %.6e, 0, %.6e)', tb, tb + Hf);
    % Fin top
    L{end+1} = sprintf('hi_addsegment(0, %.6e, %.6e, %.6e)', tb + Hf, half_tf, tb + Hf);
    % Fin right (channel wall)
    L{end+1} = sprintf('hi_addsegment(%.6e, %.6e, %.6e, %.6e)', half_tf, tb, half_tf, tb + Hf);
    L{end+1} = '';

    % Block labels
    L{end+1} = '-- Block labels';
    % Base block
    L{end+1} = sprintf('hi_addblocklabel(%.6e, %.6e)', W/2, tb/2);
    L{end+1} = sprintf('hi_selectlabel(%.6e, %.6e)', W/2, tb/2);
    L{end+1} = 'hi_setblockprop("heatsink_mat",0,0.05,0)';
    L{end+1} = 'hi_clearselected()';
    % Fin block
    L{end+1} = sprintf('hi_addblocklabel(%.6e, %.6e)', half_tf/2, tb + Hf/2);
    L{end+1} = sprintf('hi_selectlabel(%.6e, %.6e)', half_tf/2, tb + Hf/2);
    L{end+1} = 'hi_setblockprop("heatsink_mat",0,0.05,0)';
    L{end+1} = 'hi_clearselected()';
    L{end+1} = '';

    % Apply BCs
    L{end+1} = '-- Apply boundary conditions';
    % Heat flux on base bottom
    L{end+1} = sprintf('hi_selectsegment(%.6e, 0)', W/2);
    L{end+1} = 'hi_setsegmentprop("heat_flux",0,1,0,0,"<None>")';
    L{end+1} = 'hi_clearselected()';
    % Convection on fin right (channel wall)
    L{end+1} = sprintf('hi_selectsegment(%.6e, %.6e)', half_tf, tb + Hf/2);
    L{end+1} = 'hi_setsegmentprop("convection",0,1,0,0,"<None>")';
    L{end+1} = 'hi_clearselected()';
    % Convection on channel top (base top, channel portion)
    L{end+1} = sprintf('hi_selectsegment(%.6e, %.6e)', (half_tf + W)/2, tb);
    L{end+1} = 'hi_setsegmentprop("convection",0,1,0,0,"<None>")';
    L{end+1} = 'hi_clearselected()';
    % Convection on fin top
    L{end+1} = sprintf('hi_selectsegment(%.6e, %.6e)', half_tf/2, tb + Hf);
    L{end+1} = 'hi_setsegmentprop("convection",0,1,0,0,"<None>")';
    L{end+1} = 'hi_clearselected()';
    % Symmetry on left (fin + base)
    L{end+1} = sprintf('hi_selectsegment(0, %.6e)', tb/2);
    L{end+1} = 'hi_setsegmentprop("symmetry",0,1,0,0,"<None>")';
    L{end+1} = 'hi_clearselected()';
    L{end+1} = sprintf('hi_selectsegment(0, %.6e)', tb + Hf/2);
    L{end+1} = 'hi_setsegmentprop("symmetry",0,1,0,0,"<None>")';
    L{end+1} = 'hi_clearselected()';
    % Symmetry on right (base)
    L{end+1} = sprintf('hi_selectsegment(%.6e, %.6e)', W, tb/2);
    L{end+1} = 'hi_setsegmentprop("symmetry",0,1,0,0,"<None>")';
    L{end+1} = 'hi_clearselected()';
    L{end+1} = '';

    % Solve
    L{end+1} = '-- Solve';
    L{end+1} = 'hi_zoomnatural()';
    L{end+1} = 'hi_saveas("extruded_fin_model.feh")';
    L{end+1} = 'hi_analyze()';
    L{end+1} = 'hi_loadsolution()';
    L{end+1} = '';

    % Post-processing
    L{end+1} = '-- Extract results to CSV';
    L{end+1} = 'f = openfile("femm_results.csv", "w")';
    L{end+1} = 'write(f, "point,x,y,temperature\\n")';
    L{end+1} = sprintf('T_base_center = ho_getpointvalues(0, 0)');
    L{end+1} = 'write(f, string.format("base_center,0,0,%.6f\\n", T_base_center))';
    L{end+1} = sprintf('T_fin_tip = ho_getpointvalues(0, %.6e)', tb + Hf);
    L{end+1} = sprintf('write(f, string.format("fin_tip,0,%.6e,%%.6f\\n", T_fin_tip))', tb + Hf);
    L{end+1} = sprintf('T_base_top = ho_getpointvalues(0, %.6e)', tb);
    L{end+1} = sprintf('write(f, string.format("base_top,0,%.6e,%%.6f\\n", T_base_top))', tb);
    L{end+1} = 'closefile(f)';
    L{end+1} = '';
```

```matlab
function lua_str = femm_baseplate_spreading(cfg)
    % femm_baseplate_spreading - generate FEMM Lua script for baseplate spreading
    % 2D planar model: rectangular plate with heat sources on top, convection on bottom

    L = {};
    L{end+1} = '-------------------------------------------------------';
    L{end+1} = '-- Auto-generated by thermal_cli.m gen-femm';
    L{end+1} = '-- Model: baseplate-spreading (2D planar)';
    L{end+1} = sprintf('-- Date: %s', datestr(now, 'yyyy-mm-dd'));
    L{end+1} = '-------------------------------------------------------';
    L{end+1} = '';
    L{end+1} = 'newdocument(2)';
    L{end+1} = 'hi_probdef("meters","planar",1e-8,20,30)';
    L{end+1} = '';

    % Geometry
    a = cfg.heatsink.width;   % plate width [m]
    tb = cfg.heatsink.baseThickness;  % plate thickness [m]
    k = cfg.heatsink.kPlate;  % conductivity [W/(m*K)]

    % Convection
    h_bot = cfg.convection.h;
    T_fluid = cfg.convection.tFluid;

    % Sources
    p_n = cfg.sources.power;
    x_g = cfg.sources.x;
    widths = cfg.sources.width;
    n_src = length(p_n);

    L{end+1} = '-- Materials';
    L{end+1} = sprintf('hi_addmaterial("plate_mat",%.6g,%.6g,0)', k, k);
    L{end+1} = '';

    L{end+1} = '-- Boundary conditions';
    L{end+1} = sprintf('hi_addboundprop("convection_bot",2,0,0,%.6f,%.6f)', T_fluid, h_bot);
    L{end+1} = 'hi_addboundprop("insulated",1,0,0)';
    for i = 1:n_src
        q_flux = p_n(i) / (widths(i) * 1);  % per unit depth
        L{end+1} = sprintf('hi_addboundprop("source_%d",1,0,%.6e)', i, -q_flux);
    end
    L{end+1} = '';

    % Nodes: plate rectangle with source subdivisions on top
    L{end+1} = '-- Geometry nodes';
    % Bottom corners
    L{end+1} = sprintf('hi_addnode(0, 0)');
    L{end+1} = sprintf('hi_addnode(%.6e, 0)', a);
    % Top corners
    L{end+1} = sprintf('hi_addnode(0, %.6e)', tb);
    L{end+1} = sprintf('hi_addnode(%.6e, %.6e)', a, tb);

    % Source boundaries on top surface
    top_x = [0];
    for i = 1:n_src
        x_left = x_g(i) - widths(i)/2;
        x_right = x_g(i) + widths(i)/2;
        if x_left > 0
            top_x(end+1) = x_left;
        end
        top_x(end+1) = x_right;
        L{end+1} = sprintf('hi_addnode(%.6e, %.6e)', x_left, tb);
        L{end+1} = sprintf('hi_addnode(%.6e, %.6e)', x_right, tb);
    end
    L{end+1} = '';

    % Segments
    L{end+1} = '-- Segments';
    % Bottom
    L{end+1} = sprintf('hi_addsegment(0, 0, %.6e, 0)', a);
    % Left
    L{end+1} = sprintf('hi_addsegment(0, 0, 0, %.6e)', tb);
    % Right
    L{end+1} = sprintf('hi_addsegment(%.6e, 0, %.6e, %.6e)', a, a, tb);
    % Top (split by sources)
    all_x = sort(unique([0, top_x, a]));
    for i = 1:length(all_x)-1
        L{end+1} = sprintf('hi_addsegment(%.6e, %.6e, %.6e, %.6e)', all_x(i), tb, all_x(i+1), tb);
    end
    L{end+1} = '';

    % Block label
    L{end+1} = '-- Block label';
    L{end+1} = sprintf('hi_addblocklabel(%.6e, %.6e)', a/2, tb/2);
    L{end+1} = sprintf('hi_selectlabel(%.6e, %.6e)', a/2, tb/2);
    L{end+1} = 'hi_setblockprop("plate_mat",0,0.05,0)';
    L{end+1} = 'hi_clearselected()';
    L{end+1} = '';

    % Apply BCs
    L{end+1} = '-- Apply boundary conditions';
    % Bottom: convection
    L{end+1} = sprintf('hi_selectsegment(%.6e, 0)', a/2);
    L{end+1} = 'hi_setsegmentprop("convection_bot",0,1,0,0,"<None>")';
    L{end+1} = 'hi_clearselected()';
    % Left and right: insulated
    L{end+1} = sprintf('hi_selectsegment(0, %.6e)', tb/2);
    L{end+1} = 'hi_setsegmentprop("insulated",0,1,0,0,"<None>")';
    L{end+1} = 'hi_clearselected()';
    L{end+1} = sprintf('hi_selectsegment(%.6e, %.6e)', a, tb/2);
    L{end+1} = 'hi_setsegmentprop("insulated",0,1,0,0,"<None>")';
    L{end+1} = 'hi_clearselected()';
    % Top: sources and insulated gaps
    for i = 1:n_src
        x_mid = x_g(i);
        L{end+1} = sprintf('hi_selectsegment(%.6e, %.6e)', x_mid, tb);
        L{end+1} = sprintf('hi_setsegmentprop("source_%d",0,1,0,0,"<None>")', i);
        L{end+1} = 'hi_clearselected()';
    end
    L{end+1} = '';

    % Solve
    L{end+1} = '-- Solve';
    L{end+1} = 'hi_zoomnatural()';
    L{end+1} = 'hi_saveas("baseplate_model.feh")';
    L{end+1} = 'hi_analyze()';
    L{end+1} = 'hi_loadsolution()';
    L{end+1} = '';

    % Post-processing
    L{end+1} = '-- Extract results to CSV';
    L{end+1} = 'f = openfile("femm_results.csv", "w")';
    L{end+1} = 'write(f, "point,x,y,temperature\\n")';
    for i = 1:n_src
        L{end+1} = sprintf('T_%d = ho_getpointvalues(%.6e, %.6e)', i, x_g(i), tb);
        L{end+1} = sprintf('write(f, string.format("source_%d,%.6e,%.6e,%%.6f\\n", T_%d))', i, x_g(i), tb, i);
    end
    L{end+1} = sprintf('T_center_bot = ho_getpointvalues(%.6e, 0)', a/2);
    L{end+1} = sprintf('write(f, string.format("center_bottom,%.6e,0,%%.6f\\n", T_center_bot))', a/2);
    L{end+1} = 'closefile(f)';
    L{end+1} = '';
    L{end+1} = 'print("FEMM simulation complete. Results in femm_results.csv")';

    lua_str = strjoin(L, '\n');
end
```

```matlab
function result = cmd_gen_femm(parsed)
    % cmd_gen_femm - generate FEMM Lua verification script
    % Usage: thermal_cli.m gen-femm --model <type> --config <file> --output <lua_file>

    if isfield(parsed, 'help') && parsed.help
        fprintf('Usage: thermal_cli.m gen-femm --model <type> --config <file> --output <lua_file>\n');
        fprintf('Models: semi-on-pcb, extruded-fin, baseplate\n');
        result = struct();
        return;
    end

    if ~isfield(parsed, 'model')
        fprintf(2, 'Error: --model is required (semi-on-pcb|extruded-fin|baseplate)\n');
        result = struct();
        return;
    end

    cfg = cli_load_config(parsed);

    switch parsed.model
        case 'semi-on-pcb'
            lua_str = femm_semi_on_pcb(cfg);
        case 'extruded-fin'
            lua_str = femm_extruded_fin(cfg);
        case 'baseplate'
            lua_str = femm_baseplate_spreading(cfg);
        otherwise
            fprintf(2, 'Error: unknown model "%s"\n', parsed.model);
            result = struct();
            return;
    end

    if isfield(parsed, 'output')
        fid = fopen(parsed.output, 'w');
        fprintf(fid, '%s', lua_str);
        fclose(fid);
        fprintf('FEMM Lua script written to: %s\n', parsed.output);
    else
        fprintf('%s\n', lua_str);
    end

    result.lua = lua_str;
    result.model = parsed.model;
end
```

```matlab
function result = cmd_compare_femm(parsed)
    % cmd_compare_femm - compare FEMM results with analytical results
    % Usage: thermal_cli.m compare-femm --analytical <csv> --femm <csv>

    if isfield(parsed, 'help') && parsed.help
        fprintf('Usage: thermal_cli.m compare-femm --analytical <csv> --femm <csv>\n');
        fprintf('Compares analytical CSV (point,value,unit) with FEMM CSV (point,x,y,temperature).\n');
        result = struct();
        return;
    end

    % Read analytical CSV
    fid_a = fopen(parsed.analytical, 'r');
    header_a = fgetl(fid_a);
    analytical = struct();
    while ~feof(fid_a)
        line = fgetl(fid_a);
        if ischar(line) && ~isempty(line)
            parts = strsplit(line, ',');
            name = strtrim(parts{1});
            val = str2double(parts{2});
            analytical.(strrep(name, ' ', '_')) = val;
        end
    end
    fclose(fid_a);

    % Read FEMM CSV
    fid_f = fopen(parsed.femm, 'r');
    header_f = fgetl(fid_f);
    femm_data = struct();
    while ~feof(fid_f)
        line = fgetl(fid_f);
        if ischar(line) && ~isempty(line)
            parts = strsplit(line, ',');
            name = strtrim(parts{1});
            temp = str2double(parts{end});
            femm_data.(strrep(name, ' ', '_')) = temp;
        end
    end
    fclose(fid_f);

    % Compare matching points
    fprintf('%-25s %15s %15s %12s\n', 'Point', 'Analytical', 'FEMM', 'Deviation');
    fprintf('%s\n', repmat('-', 1, 70));

    result.comparisons = {};
    a_fields = fieldnames(analytical);
    f_fields = fieldnames(femm_data);

    for i = 1:length(f_fields)
        fname = f_fields{i};
        if isfield(femm_data, fname)
            femm_val = femm_data.(fname);
            % Try to find matching analytical point
            % Match by name: junction -> junction_temperature_1, etc.
            matched = false;
            for j = 1:length(a_fields)
                if ~isempty(strfind(a_fields{j}, fname)) || ~isempty(strfind(fname, a_fields{j}))
                    ana_val = analytical.(a_fields{j});
                    if ana_val ~= 0
                        dev = abs(femm_val - ana_val) / abs(ana_val) * 100;
                    else
                        dev = abs(femm_val - ana_val);
                    end
                    fprintf('%-25s %15.4f %15.4f %11.2f%%\n', fname, ana_val, femm_val, dev);
                    c.point = fname;
                    c.analytical = ana_val;
                    c.femm = femm_val;
                    c.deviation_pct = dev;
                    result.comparisons{end+1} = c;
                    matched = true;
                    break;
                end
            end
            if ~matched
                fprintf('%-25s %15s %15.4f %12s\n', fname, 'N/A', femm_val, '-');
            end
        end
    end
end
```
